\section{Complete C for an infinite-three reference followed by arbitrarily many fours}
Put $P_j=1-z-\cdots-z^j$, $h_j=P_j^{-1}$,
$S=z+z^2+z^3+z^4$, $B_s=\sum_{i<s}S^i$, and
\[
 R_s=B_s+S^s h_3,\qquad W=4s,\qquad A=s+4.
\]
Let $F_n=[z^n]R_s$, $G_n=\sum_{i\le n}F_i$, and
$D_n=F_n-F_{n-1}$ for $n\ge2$, with $D_0=D_1=0$.
Negative coefficients are zero. This is an INFINITE reference model,
not a finite actual arm. No general original-coefficient theorem is
claimed from this result.

\begin{theorem}
For EVERY $s\ge0$, $G$ is globally log-concave and $R_s$ has complete C:
\[
 G_kD_{N-1}\ge G_{k-1}D_N\qquad(N\ge4,\ 1\le k\le N-3).
\]
For $s\ge1$, the ordinary coefficients $F$ are LC at every centre
$s+4\le n\le4s-2$ (an empty band is allowed).
The number of fours, current rank and shift are all unbounded.
The golden residual need not be LC on this physical head; it is not
used as an induction premise.
\end{theorem}

\subsection{An exact polynomial tail and its retained last coefficients}
